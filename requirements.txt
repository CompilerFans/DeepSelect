# DeepSelect run-time and build-time dependencies.
#
# This file is an ADDITION, not a transcription: upstream
# (`deepseek-ai/DeepSelect`) declares no dependencies at all -- there is no
# requirements.txt on `upstream/main`, and its `setup.py` carries no
# `install_requires`.  So there is no upstream list to agree with, and what
# follows describes THIS tree: the MACA fork, which is its own repository and
# is also vendored inside mcDeepGEMM as `third-party/DeepSelect`.  That is why
# `tilelang` / `mctlassEx` are named as mcDeepGEMM's rather than ours: this
# operator has no dependency on either.
#
# Note the import name is not the distribution name for every one of these:
# `import tvm_ffi` is the distribution `apache-tvm-ffi` (there is no `tvm_ffi`
# distribution), and the same is true of mcDeepGEMM's own edge.  Pin
# what a resolver wants, not what `import` says.
#
# This file does NOT list the MACA toolkit (`$MACA_PATH`, default `/opt/maca`)
# or the MACA-compatible PyTorch build.  Both are external to anything pip can
# install -- they come with the SDK install -- and the extension links
# `libmcruntime` out of that toolkit.  A machine without them cannot build this
# tree regardless of what is listed below.

# ---- run time -------------------------------------------------------------
# tvm_ffi is the loader.  The extensions are built at the Apache TVM FFI ABI
# and export `__tvm_ffi_*` symbols with no `PyInit`, so they cannot be imported
# as python modules at all -- `deep_select/_binding.py` loads them with
# `tvm_ffi.load_module`.  The `import tvm_ffi` there is load-bearing beyond
# that call: the extension carries `DT_NEEDED: libtvm_ffi.so` and no rpath, and
# importing the package is what maps that library into the process.  The
# launch-time stream handoff calls `tvm_ffi.core._env_set_current_stream`
# directly rather than `tvm_ffi.use_torch_stream()` (measured 12.9 us -> 1.1 us
# per call; see `_binding.launching()`).
#
# The floor is `0.1.2` because that is tilelang's own declared range for this
# package (`apache-tvm-ffi>=0.1.2,~=0.1.0`); `~=0.1.0` keeps the compatible-
# release window to the 0.1 line.  The API surface this tree uses is narrow --
# `load_module`, `core._env_set_current_stream` -- but pinning to the range the
# rest of the stack already agrees on costs nothing.
apache-tvm-ffi>=0.1.2,~=0.1.0

# torch is required at the call site, not by the extension.  Tensors cross the
# ABI as DLPack and output buffers are allocated with `torch.empty`, so a
# `torch.Tensor` is what has to be on the outside of the boundary even though
# the extension no longer links libtorch/libc10.  Bare `import deep_select`
# touches torch immediately (`deep_select/interface.py`), so it is not optional
# for any use of this package.
#
# The floor matches mcDeepGEMM's own `install_requires`.  The real
# constraint is narrower and not expressible here: it must be the
# MACA-compatible PyTorch for the toolkit on the machine, and a wheel built
# against one torch major/minor must be loaded by that same one.
torch>=2.1.0

# ---- tests ----------------------------------------------------------------
# Not needed to run `deep_select.topk`; needed by the suites under `tests/` and
# `scripts/`, which is how this tree is validated.
#
# `tests/kernelkit/__init__.py` imports `bench`, and `bench` imports triton at
# module scope (the profiler-range marker kernel is `@triton.jit`), so
# `import kernelkit` -- and therefore `tests/test.py`, `tests/lib.py` and
# `scripts/perf_snapshot.py`, all of which import it -- raises ImportError
# without triton.  Note that this includes the *correctness* suite, which never
# times anything: the import is module-scope, so it is collateral.
#
# The marker is upstream's own scaffolding, not this tree's -- no kernel here
# is triton.  `bench.py`'s `KINETO_LOG_LEVEL` suppression is CUDA-side and is
# carried here; the Ascend-only profiler surface upstream added later is not.
# The rest of the `tests/` delta is recorded in CLAUDE.md: the
# `--backend` / `--seed` / `--sample` surface and the status loop in
# `test.py`, the checked `check_result` extraction, `platform.py`'s device
# probe, `kernelkit/build.py`'s MACA stack check, and two UInt workarounds in
# `lib.py`.
triton

# `numpy` is used in exactly one place, `tests/kernelkit/utils.py`'s
# `set_random_seed`, and imported lazily inside the function.  It is not
# transitively guaranteed on a torch-only install, so it is listed rather than
# assumed.
numpy

# ---- build only -----------------------------------------------------------
# `build.sh` and `setup.py` are setuptools-driven, with torch's
# `BuildExtension.with_options(use_ninja=True)` as the backend.  Neither has a
# version floor this tree depends on, so none is stated.
#
# `ninja` is a fallback, not a hard requirement: torch warns and drops to the
# slow distutils backend when `is_ninja_available()` is false.  It is listed
# because the fallback is a build this tree has never been tested on.
setuptools
ninja

# `install.sh` builds a wheel, and `bdist_wheel` moved into setuptools -- there
# is no separate `wheel` requirement any more.  It is NOT listed for that
# reason; add it only if an old setuptools on the target needs it.

# ---- NOT listed, and why --------------------------------------------------
# `ray`: no longer relevant -- upstream deleted the Ray-based stress harness
# (`tests/kernelkit/stress.py`) in `bfa4507`, and this tree no longer carries
# the file either.
#
# `tilelang` / `mctlassEx`: mcDeepGEMM's dependencies, not this one's.
# No file in this tree imports either.
#
# `pytest`: there is no pytest suite here and no conftest.py -- unlike
